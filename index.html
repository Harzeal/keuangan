Cara Mengaktifkan Database Supabase (Hanya 1 Kali):

  1. Buka dashboard Supabase Anda di SQL Editor Supabase.
  2. Klik tombol "Salin Kode SQL" di bawah ini.
  3. Tempelkan (Paste) ke editor Supabase, lalu klik tombol hijau "RUN" di pojok kanan bawah.
  4. Kembali ke halaman ini dan klik tombol "Tes Koneksi Ulang". Selesai! Data Anda akan otomatis tersimpan di cloud.
supabase_schema.sql 
-- 1. Buat Tabel Transaksi
CREATE TABLE IF NOT EXISTS public.transactions (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    type TEXT NOT NULL CHECK (type IN ('income', 'expense')),
    amount NUMERIC(15, 2) NOT NULL CHECK (amount > 0),
    title TEXT NOT NULL,
    category TEXT NOT NULL DEFAULT 'Lainnya',
    date DATE NOT NULL DEFAULT CURRENT_DATE,
    notes TEXT,
    created_at TIMESTAMPTZ NOT NULL DEFAULT timezone('utc'::text, now())
);

-- 2. Buat Tabel Anggaran Bulanan
CREATE TABLE IF NOT EXISTS public.monthly_budgets (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    month_year TEXT NOT NULL UNIQUE,
    budget_amount NUMERIC(15, 2) NOT NULL DEFAULT 0,
    notes TEXT,
    updated_at TIMESTAMPTZ NOT NULL DEFAULT timezone('utc'::text, now())
);

-- 3. Kebijakan Akses (RLS) untuk Publishable Key
ALTER TABLE public.transactions ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.monthly_budgets ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Public full access transactions" 
ON public.transactions FOR ALL TO anon, authenticated USING (true) WITH CHECK (true);

CREATE POLICY "Public full access monthly_budgets" 
ON public.monthly_budgets FOR ALL TO anon, authenticated USING (true) WITH CHECK (true);
